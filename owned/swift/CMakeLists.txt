cmake_minimum_required(VERSION 3.24)
project(StrataSwift LANGUAGES C CXX CUDA)
find_package(CUDAToolkit REQUIRED)
if(NOT CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.4" OR CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.5")
  message(FATAL_ERROR "The reconstructed profile is qualified for CUDA13.4; qualify another toolchain separately")
endif()
if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "89")
  message(FATAL_ERROR "This native source profile targets SM89; do not silently substitute an architecture")
endif()
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL "18" OR CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL "19")
  message(FATAL_ERROR "Use the qualified Clang18 toolchain")
endif()
list(GET CUDAToolkit_INCLUDE_DIRS 0 CUDAToolkit_INCLUDE_DIR)
find_program(STRATA_MOLD mold REQUIRED)
find_program(STRATA_AR NAMES llvm-ar ar REQUIRED)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Select Clang C/CXX and NVIDIA CUDA host compiler matching the qualified source")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/objects" "${CMAKE_CURRENT_BINARY_DIR}/archives")
file(GLOB_RECURSE STRATA_CLASSIC_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/source/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/source/*.cuh" "${CMAKE_CURRENT_SOURCE_DIR}/source/*.inc" "${CMAKE_CURRENT_SOURCE_DIR}/source/*.in" "${CMAKE_CURRENT_SOURCE_DIR}/source/*.inl" "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp")
list(APPEND STRATA_CLASSIC_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../include/strata/core/required_native_peer.hpp")
file(GLOB_RECURSE STRATA_SHARED_PEER_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../common/include/*.hpp")
list(APPEND STRATA_CLASSIC_HEADERS ${STRATA_SHARED_PEER_HEADERS})

set(STRATA_SWIFT_PROFILE "swift2-v454" CACHE STRING "Exact numerical provider profile")
set_property(CACHE STRATA_SWIFT_PROFILE PROPERTY STRINGS swift1-256k swift1-512k swift1-1024k swift2-v454 swift3-v431 swift4-v421-experimental)
if(NOT STRATA_SWIFT_PROFILE MATCHES "^(swift1-(256|512|1024)k|swift2-v454|swift3-v431|swift4-v421-experimental)$")
  message(FATAL_ERROR "Unknown exact source profile; no replacement backend is selected")
endif()
set(STRATA_PROFILE_HEADERS ${STRATA_CLASSIC_HEADERS})
include("${CMAKE_CURRENT_SOURCE_DIR}/profiles/${STRATA_SWIFT_PROFILE}.cmake")
if(STRATA_SWIFT_PROFILE STREQUAL "swift2-v454")
  set(STRATA_SELECTED_NATIVE "${STRATA_CLASSIC_BINARY}")
else()
  set(STRATA_SELECTED_NATIVE "${STRATA_PROFILE_BINARY}")
endif()
set(STRATA_OWNED_NATIVE "${CMAKE_CURRENT_BINARY_DIR}/strata-native")
add_custom_command(OUTPUT "${STRATA_OWNED_NATIVE}" "${CMAKE_CURRENT_BINARY_DIR}/native-profile.json"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${STRATA_SELECTED_NATIVE}" "${STRATA_OWNED_NATIVE}"
  COMMAND "${CMAKE_COMMAND}" "-DPROFILE=${STRATA_SWIFT_PROFILE}" "-DBINARY=${STRATA_OWNED_NATIVE}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/native-profile.json" -P "${CMAKE_CURRENT_SOURCE_DIR}/../common/write_profile.cmake"
  DEPENDS "${STRATA_SELECTED_NATIVE}" "${CMAKE_CURRENT_SOURCE_DIR}/../common/write_profile.cmake"
  VERBATIM)
add_custom_target(strata_owned_swift ALL DEPENDS "${STRATA_OWNED_NATIVE}" "${CMAKE_CURRENT_BINARY_DIR}/native-profile.json")
