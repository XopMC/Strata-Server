cmake_minimum_required(VERSION 3.24)
project(StrataServerOwned LANGUAGES C CXX)
set(STRATA_OWNED_FAMILY "swift" CACHE STRING "Native numerical family")
set_property(CACHE STRATA_OWNED_FAMILY PROPERTY STRINGS swift deepseek)
set(STRATA_OWNED_GPU_COUNT "2" CACHE STRING "Selected GPU count")
set(STRATA_OWNED_CONTEXT "524288" CACHE STRING "Context for source variants which require it")
option(STRATA_OWNED_EXPERIMENTAL "Explicitly admit preserved experimental providers" OFF)
if(STRATA_OWNED_FAMILY STREQUAL "swift")
  if(STRATA_OWNED_GPU_COUNT STREQUAL "1")
    if(STRATA_OWNED_CONTEXT STREQUAL "262144")
      set(STRATA_SWIFT_PROFILE "swift1-256k")
    elseif(STRATA_OWNED_CONTEXT STREQUAL "524288")
      set(STRATA_SWIFT_PROFILE "swift1-512k")
    elseif(STRATA_OWNED_CONTEXT STREQUAL "1048576")
      set(STRATA_SWIFT_PROFILE "swift1-1024k")
    else()
      message(FATAL_ERROR "No selected one-card source profile for this context")
    endif()
  elseif(STRATA_OWNED_GPU_COUNT STREQUAL "2")
    set(STRATA_SWIFT_PROFILE "swift2-v454")
  elseif(STRATA_OWNED_GPU_COUNT STREQUAL "3")
    set(STRATA_SWIFT_PROFILE "swift3-v431")
  elseif(STRATA_OWNED_GPU_COUNT STREQUAL "4" AND STRATA_OWNED_EXPERIMENTAL)
    set(STRATA_SWIFT_PROFILE "swift4-v421-experimental")
  else()
    message(FATAL_ERROR "No admitted source profile for this count;4cards requires explicit experimental admission,5..7remain unqualified")
  endif()
  add_subdirectory(swift)
elseif(STRATA_OWNED_FAMILY STREQUAL "deepseek" AND STRATA_OWNED_GPU_COUNT STREQUAL "2" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deepseek/CMakeLists.txt")
  set(STRATA_DS_CONTEXT "${STRATA_OWNED_CONTEXT}" CACHE STRING "DeepSeek context" FORCE)
  add_subdirectory(deepseek)
else()
  message(FATAL_ERROR "Requested numerical family/count is not source-integrated")
endif()
