cmake_minimum_required(VERSION 3.24)
project(StrataNativePeerPolicyTests LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
 message(FATAL_ERROR "Use Clang with mold")
endif()
add_executable(native_peer_policy ../../tests/native_peer_policy.cpp ../../src/core/required_native_peer.cpp)
target_include_directories(native_peer_policy PRIVATE ../../include)
target_compile_features(native_peer_policy PRIVATE cxx_std_20)
target_compile_definitions(native_peer_policy PRIVATE STRATA_NATIVE_PEER_CPU_ONLY=1)
target_link_options(native_peer_policy PRIVATE -fuse-ld=mold)
enable_testing()
add_test(NAME native_peer_policy COMMAND native_peer_policy)
